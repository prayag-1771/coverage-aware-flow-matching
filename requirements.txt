# numpy is pinned below 2 and the pin is load-bearing.
#
# pyarrow 15.0.2 is compiled against the numpy 1.x C API and fails at import with
# "AttributeError: _ARRAY_API not found" under numpy 2.x, which breaks reading the
# CICIDS2017 parquet. The pyarrow releases built for numpy 2 are blocked by Windows
# Application Control on this machine (see below), so upgrading pyarrow is not an
# option and numpy must stay below 2.
#
# `pip install shap` silently upgrades numpy to 2.x and breaks pyarrow. shap 0.51.0
# declares numpy>=2 and pip will warn that this pin violates it, but TreeSHAP was
# verified working on numpy 1.26.4 - the declared bound is stricter than the actual
# requirement. Keep the pin and ignore the warning.
numpy==1.26.4
pyarrow==15.0.2
shap==0.51.0

# PyTorch, CUDA 12.4 build. NOT resolvable from PyPI -- a plain `pip install
# torch` yields a CPU-only build, and flow matching then runs on CPU silently,
# with no error or warning. Install with:
#
#   pip install --index-url https://download.pytorch.org/whl/cu124 torch==2.6.0+cu124
#
# On a slow or unreliable connection, fetch the wheel with curl first (pip does
# not resume a failed download; curl does) and install the local file:
#
#   curl -L -C - --retry 20 --retry-all-errors \
#     -o torch-2.6.0+cu124-cp311-cp311-win_amd64.whl \
#     https://download.pytorch.org/whl/cu124/torch-2.6.0%2Bcu124-cp311-cp311-win_amd64.whl
#
# The output filename must be the canonical wheel name; pip rejects anything
# else with "Invalid wheel filename (wrong number of parts)".
#
# CPU-only environments are fine -- everything runs, just slower.
# src/models/classifier.py selects the device automatically, and every
# experiment logs which device it used.
torch==2.6.0+cu124
